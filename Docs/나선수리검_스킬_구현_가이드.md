# 나선수리검(RasenShuriken) 스킬 구현 가이드

> 작성일: 2026-04-06  
> 목적: 오른손 Attach → 노티파이 발사 → 타겟 유도 → 다단히트 전체 흐름 구현

---

## 전체 흐름 / 구조

```
[PlayerState_Skill : Rasen_Shuriken 입력]
    └─ AnimationStateComponent::Play_State("RasenShuriken")
         │
         ├─ [ANS_AttachSkill] (NotifyState 구간)
         │     ├─ On_Begin : Skill_RasenShuriken 스폰 + 오른손 뼈에 위치 동기화
         │     ├─ On_Tick  : 매 프레임 뼈 행렬로 수리검 위치/회전 갱신
         │     └─ On_End   : 미발사 상태면 Set_Destroy(true) (안전 정리)
         │
         └─ [AN_LaunchSkill] (Notify 포인트)
               ├─ TargetComponent → 락온 타겟 방향 계산
               ├─ 없으면 owner Forward 사용
               └─ Skill_RasenShuriken::Launch(direction) 호출
                     └─ ProjectileComponent 활성화 → 이동 시작
                          └─ OnBeginOverlap 다단히트 처리
```

```cpp
// Client/Public/SkillObject_Projectile.h
#pragma once

#include "SkillObject.h"

NS_BEGIN(Engine)
class ProjectileComponent;
NS_END

NS_BEGIN(Client)

// SkillObject에 ProjectileComponent를 붙여 날아가는 투사체 기반 클래스.
// startAttached = true로 초기화하면 Launch() 호출 전까지 이동하지 않는다.
// (오른손에 붙어있는 수리검 차지 구간에 사용)
class SkillObject_Projectile : public SkillObject
{
    GENERATED_BODY(SkillObject_Projectile)

public:
    struct FProjectileSkillDesc : public SkillObject::FSkillObjectDesc
    {
        Vec3    direction    = Vec3::Forward; // 초기 발사 방향 (정규화 필수)
        float   speed        = 20.f;         // 이동 속도 (units/sec)
        float   maxDistance  = 50.f;         // 최대 사거리
        bool    useGravity   = false;        // 중력 영향 여부

        // true면 Launch() 호출 전까지 ProjectileComponent가 이동하지 않는다.
        // 오른손 Attach 상태에서 수리검이 제자리에 머무르게 할 때 사용.
        bool    startAttached = false;
    };

public:
    explicit SkillObject_Projectile(ComPtr<Device> device, ComPtr<DeviceContext> context);
    explicit SkillObject_Projectile(const SkillObject_Projectile& rhs);
    virtual ~SkillObject_Projectile() = default;

public:
    HRESULT Initialize_Prototype() override;
    HRESULT Initialize(void* arg) override;
    void    Update(float timeDelta) override;

    void    OnBeginOverlap(Shared<Collider> self, Shared<Collider> other) override;

public:
    // 외부(AN_LaunchSkill)에서 호출 — 방향 세팅 후 이동 활성화.
    // 이 함수가 호출되기 전까지 ProjectileComponent는 정지 상태.
    virtual void Launch(const Vec3& direction);

    // 현재 발사 상태인지 외부에서 체크.
    bool IsLaunched() const { return _isMoving; }

protected:
    // ProjectileComponent 인스턴스 (Add_Component로 생성)
    Shared<ProjectileComponent> _projectile;

    // false이면 Update에서 Update_Projectile() 호출 안 함.
    // Launch() 호출 시 true로 전환됨.
    bool _isMoving = true;

public:
    static Shared<GameObject> Create(ComPtr<Device> device, ComPtr<DeviceContext> context);
    Shared<GameObject> Clone(void* arg) override;
    void Free() override;
};

NS_END
```

```cpp
#include "pch.h"
#include "SkillObject_Projectile.h"

#include "Collider.h"

REGISTER_GAMEOBJECT(SkillObject_Projectile, Protocol::OBJECT_TYPE_SKILL_PROJECTILE)

SkillObject_Projectile::SkillObject_Projectile(ComPtr<Device> device, ComPtr<DeviceContext> context)
    : SkillObject(device, context)
{
}

SkillObject_Projectile::SkillObject_Projectile(const SkillObject_Projectile& rhs)
    : SkillObject(rhs)
{
}

HRESULT SkillObject_Projectile::Initialize_Prototype()
{
    return SkillObject::Initialize_Prototype();
}

HRESULT SkillObject_Projectile::Initialize(void* arg)
{
    CHECK_FAILED(SkillObject::Initialize(arg), E_FAIL);

    auto* desc = static_cast<FProjectileSkillDesc*>(arg);
    if (!desc)
        return E_FAIL;

    // startAttached = true이면 초기에는 정지 상태.
    // Launch() 호출 시 _isMoving = true로 전환됨.
    _isMoving = !desc->startAttached;

    // ProjectileComponent 초기화 (방향/속도 세팅, 현재 이동 여부와 무관하게 생성)
    ProjectileComponent::FProjectileDesc projDesc;
    projDesc.direction   = desc->direction;
    projDesc.speed       = desc->speed;
    projDesc.maxDistance = desc->maxDistance;
    projDesc.maxLifetime = desc->lifetime;
    projDesc.useGravity  = desc->useGravity;

    CHECK_FAILED(Add_Component(Protocol::COMPONENT_TYPE_PROJECTILE, _projectile, &projDesc), E_FAIL);

    return S_OK;
}

void SkillObject_Projectile::Update(float timeDelta)
{
    // 발사 상태일 때만 이동 처리
    if (_isMoving && _projectile)
        _projectile->Update_Projectile(timeDelta);

    // 수명 체크
    SkillObject::Update(timeDelta);
}

void SkillObject_Projectile::OnBeginOverlap(Shared<Collider> self, Shared<Collider> other)
{
    SkillObject::OnBeginOverlap(self, other);

    if (!other || Is_Destroy())
        return;

    auto otherOwner = other->Get_Owner();
    if (!otherOwner)
        return;

    const auto objectType = otherOwner->Get_ObjectType();

    // 기본 동작: 몬스터 충돌 시 파괴.
    // Skill_RasenShuriken이 이 함수를 오버라이드해서 다단히트 처리함.
    if (objectType == Protocol::OBJECT_TYPE_MONSTER)
    {
        Set_Destroy(true);
    }
}

void SkillObject_Projectile::Launch(const Vec3& direction)
{
    if (_isMoving)
        return; // 이미 발사 중이면 무시

    if (_projectile)
        _projectile->Set_Direction(direction);

    _isMoving = true;
}

Shared<GameObject> SkillObject_Projectile::Create(ComPtr<Device> device, ComPtr<DeviceContext> context)
{
    auto instance = make_shared<SkillObject_Projectile>(device, context);

    if (FAILED(instance->Initialize_Prototype()))
    {
        MSG_BOX("Failed to Create : SkillObject_Projectile");
        return nullptr;
    }

    return instance;
}

Shared<GameObject> SkillObject_Projectile::Clone(void* arg)
{
    auto clone = make_shared<SkillObject_Projectile>(*this);

    if (FAILED(clone->Initialize(arg)))
    {
        MSG_BOX("Failed to Clone : SkillObject_Projectile");
        return nullptr;
    }

    return clone;
}

void SkillObject_Projectile::Free()
{
    SkillObject::Free();
}
```

```cpp
// Client/Public/Skill_RasenShuriken.h
#pragma once

#include "SkillObject_Projectile.h"

NS_BEGIN(Client)

// 나선수리검 스킬 오브젝트.
// ANS_AttachSkill이 오른손 뼈에 붙여주고 위치를 동기화한다.
// AN_LaunchSkill이 Launch()를 호출해 실제 발사를 시작한다.
// 발사 후 OnBeginOverlap에서 다단히트를 처리한다.
class Skill_RasenShuriken : public SkillObject_Projectile
{
    GENERATED_BODY(Skill_RasenShuriken)

public:
    explicit Skill_RasenShuriken(ComPtr<Device> device, ComPtr<DeviceContext> context);
    explicit Skill_RasenShuriken(const Skill_RasenShuriken& rhs);
    virtual ~Skill_RasenShuriken() = default;

public:
    HRESULT Initialize_Prototype() override;
    HRESULT Initialize(void* arg) override;
    void    Update(float timeDelta) override;

    // 충돌 시 다단히트 처리 (SkillObject_Projectile의 기본 즉시파괴를 대체)
    void    OnBeginOverlap(Shared<Collider> self, Shared<Collider> other) override;

public:
    // ANS_AttachSkill의 On_Tick에서 매 프레임 호출.
    // 뼈의 월드 행렬을 그대로 받아서 수리검 위치/회전을 맞춰준다.
    void    Sync_AttachedTransform(const Matrix& boneWorldMatrix);

    // Launch 오버라이드 — 발사 시 콜라이더도 활성화
    void    Launch(const Vec3& direction) override;

private:
    // 다단히트: 한 번 히트할 때마다 카운트 증가.
    // _maxHitCount 도달 시 수리검 파괴.
    int32   _hitCount    = 0;
    int32   _maxHitCount = 5;    // 최대 히트 횟수 (조정 가능)

    // 같은 대상에 너무 빨리 재히트하지 않도록 per-target 쿨타임
    float   _hitInterval = 0.12f;                    // 히트 간격 (초)
    umap<GameObject*, float> _hitCooldowns;          // target → 마지막 히트 시각

    // 히트 시 대상 Monster의 MovementComponent에 전달할 Launch 힘
    float   _hitLaunchForce = 280.f;

    // 수리검 고유 파라미터 (Initialize에서 desc에 덮어씀)
    float   _speed          = 28.f;
    float   _maxDistance    = 35.f;
    float   _lifetime_sec   = 2.5f;
    float   _colliderRadius = 1.35f;

public:
    static Shared<GameObject> Create(ComPtr<Device> device, ComPtr<DeviceContext> context);
    Shared<GameObject> Clone(void* arg) override;
    void Free() override;
};

NS_END
```

```cpp
#include "Monster.h"

REGISTER_GAMEOBJECT_CATEGORY(Skill_RasenShuriken, Protocol::OBJECT_TYPE_SKILL_RASENSHURIKEN, "SkillSpawn");

Skill_RasenShuriken::Skill_RasenShuriken(ComPtr<Device> device, ComPtr<DeviceContext> context)
    : SkillObject_Projectile(device, context)

    : SkillObject_Projectile(rhs)

HRESULT Skill_RasenShuriken::Initialize_Prototype()
{
    return SkillObject_Projectile::Initialize_Prototype();
}

HRESULT Skill_RasenShuriken::Initialize(void* arg)
{
    auto* srcDesc = static_cast<SkillObject_Projectile::FProjectileSkillDesc*>(arg);
    if (!srcDesc)
        return E_FAIL;

    // 외부 desc를 복사한 뒤 수리검 고유 파라미터로 덮어씀
    SkillObject_Projectile::FProjectileSkillDesc desc = *srcDesc;

    desc.colliderType   = Protocol::COMPONENT_TYPE_COLLIDER_SPHERE;
    desc.colliderRadius = _colliderRadius;
    desc.speed          = _speed;
    desc.maxDistance    = _maxDistance;
    desc.lifetime       = _lifetime_sec;
    desc.useGravity     = false;
    desc.scale          = Vec3(1.f, 1.f, 1.f);

    // ANS_AttachSkill이 오른손에 붙이는 동안 이동하지 않아야 함.
    // AN_LaunchSkill이 호출하는 Launch()에서 이동 시작됨.
    desc.startAttached = true;

    CHECK_FAILED(SkillObject_Projectile::Initialize(&desc), E_FAIL);

    // 초기에는 콜라이더 비활성 — Launch() 시 활성화
    if (_collider)
        _collider->Set_Active(false);

    return S_OK;
}

void Skill_RasenShuriken::Update(float timeDelta)
{
    SkillObject_Projectile::Update(timeDelta);

    if (Is_Destroy())
        return;

    // 히트 쿨타임 감소 갱신
    for (auto& [target, remainTime] : _hitCooldowns)
        remainTime -= timeDelta;
}

void Skill_RasenShuriken::Launch(const Vec3& direction)
{
    // 부모의 Launch 호출 → _isMoving = true + 방향 세팅
    SkillObject_Projectile::Launch(direction);

    // 발사 시 콜라이더 활성화
    if (_collider)
        _collider->Set_Active(true);
}

void Skill_RasenShuriken::Sync_AttachedTransform(const Matrix& boneWorldMatrix)
{
    // 발사 후에는 호출해도 위치를 덮지 않음
    if (IsLaunched())
        return;

    if (!_transformCom)
        return;

    // 뼈의 월드 행렬에서 위치/회전/스케일을 그대로 반영
    Vec3 position, scale;
    Quaternion rotation;
    boneWorldMatrix.Decompose(scale, rotation, position);

    _transformCom->Set_LocalPosition(position);
    _transformCom->Set_LocalQuaternion(rotation);
    // 수리검 자체 스케일은 유지 (뼈 스케일 무시)
}

void Skill_RasenShuriken::OnBeginOverlap(Shared<Collider> self, Shared<Collider> other)
{
    // 발사 전이면 히트 판정 안 함
    if (!IsLaunched())
        return;

    if (!other || Is_Destroy())
        return;

    auto otherOwner = other->Get_Owner();
    if (!otherOwner)
        return;

    // 몬스터에만 히트 처리
    if (otherOwner->Get_ObjectType() != Protocol::OBJECT_TYPE_MONSTER)
        return;

    // per-target 쿨타임 체크
    auto it = _hitCooldowns.find(otherOwner.get());
    if (it != _hitCooldowns.end() && it->second > 0.f)
        return; // 아직 쿨타임 중

    // 최대 히트 횟수 체크
    if (_hitCount >= _maxHitCount)
    {
        Set_Destroy(true);
        return;
    }

    // 히트 처리
    ++_hitCount;
    _hitCooldowns[otherOwner.get()] = _hitInterval;

    auto monster = dynamic_cast<Monster*>(otherOwner.get());
    if (!monster)
        return;

    // 데미지 이벤트 구성
    FDamageEvent dmgEvent;
    dmgEvent.damage   = 30.f;              // TODO : 스킬 데이터 테이블 연동
    dmgEvent.attacker = nullptr;           // TODO : 발사한 플레이어 weak 참조
    monster->TakeDamage(dmgEvent);

    // Launch(공중 뜨기) 적용: 위 방향 + 약간 뒤쪽으로
    auto monsterMovement = monster->Get_Component<MovementComponent>();
    if (monsterMovement)
    {
        Vec3 launchDir = Vec3(0.f, 1.f, 0.f); // 위 방향
        monsterMovement->Launch(launchDir * _hitLaunchForce, false, true);
    }

    // 마지막 히트에서 파괴
    if (_hitCount >= _maxHitCount)
        Set_Destroy(true);
}

Shared<GameObject> Skill_RasenShuriken::Create(ComPtr<Device> device, ComPtr<DeviceContext> context)
{
    auto instance = make_shared<Skill_RasenShuriken>(device, context);

    if (FAILED(instance->Initialize_Prototype()))
    {
        MSG_BOX("Failed to Create : Skill_RasenShuriken");
        return nullptr;
    }

    return instance;
}

Shared<GameObject> Skill_RasenShuriken::Clone(void* arg)
{
    auto clone = make_shared<Skill_RasenShuriken>(*this);

    if (FAILED(clone->Initialize(arg)))
    {
        MSG_BOX("Failed to Clone : Skill_RasenShuriken");
        return nullptr;
    }

    return clone;
```

```cpp
// Client/Public/ANS_AttachSkill.h
#pragma once

#include "AnimNotifyState.h"

NS_BEGIN(Client)

class Skill_RasenShuriken;

// 애니메이션 구간 동안 스킬 오브젝트를 특정 본(소켓)에 Attach하는 AnimNotifyState.
//
// On_Begin : 스킬 오브젝트를 스폰하고 약한 참조(_attachedSkill)로 보관.
//            초기에는 콜라이더 비활성 상태 (Skill_RasenShuriken::Initialize 내부에서 처리).
// On_Tick  : 매 프레임 model->Get_SocketBoneMatrixPtr()로 뼈 행렬을 가져와
//            스킬 오브젝트의 위치/회전을 동기화.
// On_End   : 아직 발사되지 않은 상태면 스킬 오브젝트 파괴 (안전 정리).
//            이미 발사된 상태면 아무것도 하지 않음.
class ANS_AttachSkill : public AnimNotifyState
{
    GENERATED_BODY(ANS_AttachSkill)

public:
    string Get_TypeName() const override;

    void On_Begin(const FAnimNotifyContext& context) override;
    void On_Tick(const FAnimNotifyContext& context)  override;
    void On_End(const FAnimNotifyContext& context)   override;

public:
    json Serialize_Payload() const override;
    void Deserialize_Payload(const json& payload) override;

private:
    // 에디터에서 세팅하는 스폰 파라미터
    Protocol::OBJECT_TYPE _spawnObjectType = Protocol::OBJECT_TYPE_SKILL_RASENSHURIKEN;
    string                _boneName        = "hand_r";          // 붙일 뼈 이름 (에디터 조정)
    string                _layerTag        = "Layer_SkillObject"; // 스폰할 레이어 태그
    int32                 _skill_Id        = 0;                  // 스킬 구분 ID

    // On_Begin에서 스폰한 수리검의 약한 참조.
    // Tick에서 위치 동기화하고, End에서 미발사 시 정리하는 데 사용.
    Weak<Skill_RasenShuriken> _attachedSkill;
};

NS_END
```

```cpp
bool ANS_AttachSkill::Register_Properties()
{
    auto& info = GetStaticReflectionInfo();
    info.className = "ANS_AttachSkill";
    info.properties.clear();

    PROPERTY_INT("스킬 ID", _skill_Id, 0, 9999);
    PROPERTY_STRING("뼈 이름", _boneName);
    PROPERTY_STRING("레이어 태그", _layerTag);

string ANS_AttachSkill::Get_TypeName() const
{
    return "ANS_AttachSkill";
}

void ANS_AttachSkill::On_Begin(const FAnimNotifyContext& context)
{
    if (context.isPreview)
        return;

    if (!context.owner)
        return;

    // 스킬 오브젝트 스폰 (startAttached = true 이므로 정지 상태로 생성됨)
    SkillObject_Projectile::FProjectileSkillDesc desc;
    desc.ownerSkillId   = _skill_Id;
    desc.spawnPosition  = Vec3::Zero; // 곧바로 Tick에서 뼈 위치로 덮어씀
    desc.spawnRotation  = Vec3::Zero;
    desc.scale          = Vec3::One;
    desc.direction      = Vec3::Forward; // 임시 — Launch 시 덮어씀
    desc.startAttached  = true;          // 이동 잠금

    const uint32  currentLevel = GAME->Current_Level();
    const wstring layerTagW    = Utils::ToWString(_layerTag);

    auto spawned = GAME->Clone_And_Add_GameObject(
        ETOI(ELevelType::Static), // 프로토타입이 등록된 레벨
        _spawnObjectType,
        currentLevel,
        layerTagW,
        &desc);

    if (!spawned)
    {
        LOG_WARN("ANS_AttachSkill::On_Begin - 스킬 오브젝트 스폰 실패. type={}, id={}",
            static_cast<int32>(_spawnObjectType), _skill_Id);
        return;
    }

    // Skill_RasenShuriken 타입으로 캐스팅해서 약한 참조 보관
    _attachedSkill = dynamic_pointer_cast<Skill_RasenShuriken>(spawned);
}

void ANS_AttachSkill::On_Tick(const FAnimNotifyContext& context)
{
    if (context.isPreview)
        return;

    auto skill = _attachedSkill.lock();
    if (!skill || skill->Is_Destroy())
        return;

    // 이미 발사된 상태면 위치 동기화 중단 (수리검이 자유비행 중)
    if (skill->IsLaunched())
        return;

    // owner 모델에서 뼈 월드 행렬 가져오기
    if (!context.model)
        return;

    const Matrix* boneMatrix = context.model->Get_SocketBoneMatrixPtr(_boneName);
    if (!boneMatrix)
    {
        LOG_WARN("ANS_AttachSkill::On_Tick - 뼈를 찾지 못함: \"{}\"", _boneName);
        return;
    }

    // owner Transform의 월드 행렬과 결합 (뼈 행렬은 로컬이므로 곱해줘야 함)
    auto ownerTransform = context.owner->Get_Component<Transform>();
    if (!ownerTransform)
        return;

    Matrix boneWorldMatrix = (*boneMatrix) * ownerTransform->Get_WorldMatrix();

    // 수리검 위치/회전 동기화
    skill->Sync_AttachedTransform(boneWorldMatrix);
}

void ANS_AttachSkill::On_End(const FAnimNotifyContext& context)
{
    if (context.isPreview)
        return;

    auto skill = _attachedSkill.lock();
    if (!skill)
        return;

    // 이미 발사됐으면 수리검은 알아서 날아가도록 놔둠
    if (skill->IsLaunched())
        return;

    // 미발사 상태 (예: 스킬 캔슬, 피격 인터럽트 등)이면 파괴
    skill->Set_Destroy(true);
    _attachedSkill.reset();
}

json ANS_AttachSkill::Serialize_Payload() const
{
    json j;
    j["spawn_object_type"] = string(magic_enum::enum_name(_spawnObjectType));
    j["bone_name"]         = _boneName;
    j["layer_tag"]         = _layerTag;
    j["skill_id"]          = _skill_Id;
    return j;
}

void ANS_AttachSkill::Deserialize_Payload(const json& payload)
{
    const string typeName = payload.value("spawn_object_type", string{});
    auto objectType = magic_enum::enum_cast<Protocol::OBJECT_TYPE>(typeName);
    if (objectType.has_value())
        _spawnObjectType = objectType.value();

    _boneName  = payload.value("bone_name",  string("hand_r"));
    _layerTag  = payload.value("layer_tag",  string("Layer_SkillObject"));
    _skill_Id  = payload.value("skill_id",   0);
}
```

```cpp
// Client/Public/AN_LaunchSkill.h
#pragma once

#include "AnimNotify.h"

NS_BEGIN(Client)

// 특정 애니메이션 프레임에 수리검을 손에서 분리, 타겟 방향으로 발사하는 AnimNotify.
//
// Execute 로직:
//   1. 현재 레이어에서 skill_Id에 해당하는 Skill_RasenShuriken을 탐색
//   2. owner의 TargetComponent에서 락온 타겟 유무 확인
//   3. 타겟이 있으면 타겟 방향, 없으면 owner Forward 방향으로 Launch() 호출
class AN_LaunchSkill : public AnimNotify
{
    GENERATED_BODY(AN_LaunchSkill)

public:
    string Get_TypeName() const override;
    void   Execute(const FAnimNotifyContext& context) override;

public:
    json Serialize_Payload() const override;
    void Deserialize_Payload(const json& payload) override;

private:
    // ANS_AttachSkill과 동일한 레이어 태그여야 탐색 가능
    string _layerTag  = "Layer_SkillObject";

    // 대응하는 ANS_AttachSkill과 동일한 skill_Id를 설정해야 함
    int32  _skill_Id  = 0;

    // 발사 방향에 더하는 수직 보정값 (살짝 위로 향하게 하려면 양수)
    float  _verticalBias = 0.1f;

private:
    // owner의 TargetComponent에서 락온 타겟 방향을 구하는 헬퍼.
    // 락온 타겟이 없으면 Vec3::Zero 반환.
    static Vec3 Get_TargetDirection(GameObject* owner);
};

NS_END
```

```cpp
bool AN_LaunchSkill::Register_Properties()
{
    auto& info = GetStaticReflectionInfo();
    info.className = "AN_LaunchSkill";
    info.properties.clear();

    PROPERTY_INT("스킬 ID", _skill_Id, 0, 9999);
    PROPERTY_STRING("레이어 태그", _layerTag);
    PROPERTY_FLOAT("수직 보정", _verticalBias, 0.f, 1.f);

    return true;
}

string AN_LaunchSkill::Get_TypeName() const
{
    return "AN_LaunchSkill";
}

void AN_LaunchSkill::Execute(const FAnimNotifyContext& context)
{
    if (context.isPreview)
        return;

    if (!context.owner)
        return;

    // ─────────────────────────────────────────────
    // 1. 레이어에서 Skill_RasenShuriken 탐색
    //    같은 레이어에 여러 수리검이 있을 경우 skill_Id로 구분
    // ─────────────────────────────────────────────
    const uint32  currentLevel = GAME->Current_Level();
    const wstring layerTagW    = Utils::ToWString(_layerTag);

    auto layer = GAME->Find_Layer(currentLevel, layerTagW);
    if (!layer)
    {
        LOG_WARN("AN_LaunchSkill::Execute - 레이어를 찾지 못함: \"{}\"", _layerTag);
        return;
    }

    Shared<Skill_RasenShuriken> targetSkill = nullptr;

    for (auto& obj : layer->Get_GameObjects())
    {
        auto shuriken = dynamic_pointer_cast<Skill_RasenShuriken>(obj);
        if (!shuriken || shuriken->Is_Destroy())
            continue;

        // 이미 발사된 수리검은 건너뜀
        if (shuriken->IsLaunched())
            continue;

        targetSkill = shuriken;
        break; // 첫 번째 미발사 수리검에 발사
    }

    if (!targetSkill)
    {
        LOG_WARN("AN_LaunchSkill::Execute - 발사할 Skill_RasenShuriken을 찾지 못함. skill_Id={}", _skill_Id);
        return;
    }

    // ─────────────────────────────────────────────
    // 2. 발사 방향 결정
    //    TargetComponent 락온 타겟 → 타겟 방향
    //    없으면 → owner Forward
    // ─────────────────────────────────────────────
    Vec3 launchDir = Get_TargetDirection(context.owner);

    if (launchDir.LengthSquared() < 0.1f)
    {
        // 락온 타겟 없음 → owner 정면 방향 사용
        auto ownerTransform = context.owner->Get_Component<Transform>();
        if (ownerTransform)
            launchDir = ownerTransform->Get_WorldForward();
        else
            launchDir = Vec3::Forward;
    }

    // 수직 보정 적용
    launchDir.y += _verticalBias;
    launchDir.Normalize();

    // ─────────────────────────────────────────────
    // 3. 발사!
    // ─────────────────────────────────────────────
    targetSkill->Launch(launchDir);
}

Vec3 AN_LaunchSkill::Get_TargetDirection(GameObject* owner)
{
    if (!owner)
        return Vec3::Zero;

    auto targetComp = owner->Get_Component<TargetComponent>();
    if (!targetComp || !targetComp->IsLockOn())
        return Vec3::Zero;

    auto lockedTarget = targetComp->Get_LockedTarget().lock();
    if (!lockedTarget)
        return Vec3::Zero;

    auto ownerTransform = owner->Get_Component<Transform>();
    auto targetTransform = lockedTarget->Get_Component<Transform>();
    if (!ownerTransform || !targetTransform)
        return Vec3::Zero;

    Vec3 toTarget = targetTransform->Get_WorldPosition() - ownerTransform->Get_WorldPosition();
    if (toTarget.LengthSquared() < 0.001f)
        return Vec3::Zero;

    toTarget.Normalize();
    return toTarget;
}

json AN_LaunchSkill::Serialize_Payload() const
{
    json j;
    j["layer_tag"]      = _layerTag;
    j["skill_id"]       = _skill_Id;
    j["vertical_bias"]  = _verticalBias;
    return j;
}

void AN_LaunchSkill::Deserialize_Payload(const json& payload)
{
    _layerTag      = payload.value("layer_tag",     string("Layer_SkillObject"));
    _skill_Id      = payload.value("skill_id",      0);
    _verticalBias  = payload.value("vertical_bias", 0.1f);
}
```

## 주의사항 / 체크리스트

### 에디터 세팅

- **ANS_AttachSkill** 구간: 차지 애니메이션 시작 ~ 발사 직전 프레임
- **AN_LaunchSkill** 포인트: 손에서 수리검을 던지는 정확한 프레임
- `ANS_AttachSkill._boneName` : 에디터 인스펙터에서 실제 뼈 이름 입력 (`hand_r` 등)
- `ANS_AttachSkill._skill_Id` == `AN_LaunchSkill._skill_Id` 동일하게 맞출 것

```
Client/Private/AN_LaunchSkill.cpp
```

### Model::Get_SocketBoneMatrixPtr 반환값

```cpp
const Matrix* Get_SocketBoneMatrixPtr(const string& boneName) const;
```

반환값이 `nullptr`이면 뼈 이름이 틀린 것이므로 에디터에서 이름 확인.  
뼈 행렬은 **로컬 공간** 기준이므로 `owner_WorldMatrix`와 반드시 곱해야 월드 위치가 됨.

### Collider::Set_Active

`SkillObject::Ready_Components`에서 콜라이더 생성 후 `Set_Active(false)` 가능 여부 확인.  
`Collider` 클래스에 `Set_Active(bool)` 없으면 간단히 추가하거나  
`GAME->Add_Collider` 호출 시점을 `Launch()` 이후로 늦추는 방법도 사용 가능.

### Transform::Set_LocalQuaternion

`Sync_AttachedTransform`에서 사용.  
`Transform`에 해당 함수 없으면 `Set_LocalRotation(float, float, float)` 대신  
`Matrix`에서 오일러 각도를 추출해서 대입하면 됨:

```cpp
Vec3 euler = rotation.ToEuler(); // SimpleMath 기준
_transformCom->Set_LocalRotation(
    XMConvertToDegrees(euler.x),
    XMConvertToDegrees(euler.y),
    XMConvertToDegrees(euler.z));
```
